    <script>
        // Initialize variables and localStorage
        let experiences = JSON.parse(localStorage.getItem('whyunified_experiences')) || [];
        let selectedType = '';
        let currentFilter = 'all';

        // DOM elements
        const experienceForm = document.getElementById('experienceForm');
        const experienceTypeInput = document.getElementById('experienceType');
        const ratingButtons = document.querySelectorAll('.rating-btn');
        const experiencesList = document.getElementById('experiencesList');
        const experienceCount = document.getElementById('experienceCount');
        const filterButtons = document.querySelectorAll('.filter-btn');

        // Set up rating buttons
        ratingButtons.forEach(button => {
            button.addEventListener('click', function() {
                ratingButtons.forEach(btn => btn.classList.remove('selected'));
                this.classList.add('selected');
                selectedType = this.getAttribute('data-value');
                experienceTypeInput.value = selectedType;
            });
        });

        // Set up filter buttons
        filterButtons.forEach(button => {
            button.addEventListener('click', function() {
                filterButtons.forEach(btn => btn.classList.remove('active'));
                this.classList.add('active');
                currentFilter = this.getAttribute('data-filter');
                displayExperiences();
            });
        });

        // Handle form submission
        experienceForm.addEventListener('submit', function(e) {
            e.preventDefault();
            
            const name = document.getElementById('name').value.trim();
            const email = document.getElementById('email').value.trim();
            const experienceText = document.getElementById('experience').value.trim();
            const type = experienceTypeInput.value;
            
            // Validate
            if (!experienceText) {
                alert('Please share your experience details');
                return;
            }
            
            if (!type) {
                alert('Please select your experience type (good or bad)');
                return;
            }
            
            // Create experience object
            const newExperience = {
                id: Date.now(),
                name: name || 'Anonymous',
                email: email,
                type: type,
                text: experienceText,
                date: new Date().toLocaleDateString('en-US', { 
                    year: 'numeric', 
                    month: 'short', 
                    day: 'numeric',
                    hour: '2-digit',
                    minute: '2-digit'
                })
            };
            
            // Add to array and save
            experiences.unshift(newExperience);
            saveExperiences();
            
            // Reset form
            experienceForm.reset();
            ratingButtons.forEach(btn => btn.classList.remove('selected'));
            experienceTypeInput.value = '';
            selectedType = '';
            
            // Show success message
            alert('Thank you for sharing your experience! It has been added to the website.');
            
            // Update display
            displayExperiences();
        });

        // Save experiences to localStorage
        function saveExperiences() {
            localStorage.setItem('whyunified_experiences', JSON.stringify(experiences));
        }

        // Display experiences
        function displayExperiences() {
            // Filter experiences based on current filter
            let filteredExperiences = experiences;
            if (currentFilter !== 'all') {
                filteredExperiences = experiences.filter(exp => exp.type === currentFilter);
            }
            
            // Update count
            experienceCount.textContent = `${filteredExperiences.length} Experience${filteredExperiences.length !== 1 ? 's' : ''}`;
            
            // Clear current list
            experiencesList.innerHTML = '';
            
            // If no experiences
            if (filteredExperiences.length === 0) {
                const noExp = document.createElement('div');
                noExp.className = 'no-experiences';
                noExp.innerHTML = `<p>No ${currentFilter !== 'all' ? currentFilter : ''} experiences to display. Share yours above!</p>`;
                experiencesList.appendChild(noExp);
                return;
            }
            
            // Add each experience to the list
            filteredExperiences.forEach(exp => {
                const expElement = document.createElement('div');
                expElement.className = `experience-item ${exp.type}`;
                
                const typeText = exp.type === 'good' ? 'Good Experience' : 'Bad Experience';
                const typeClass = exp.type === 'good' ? 'type-good' : 'type-bad';
                
                expElement.innerHTML = `
                    <div class="experience-header">
                        <div class="experience-name">${exp.name}</div>
                        <div class="experience-type ${typeClass}">${typeText}</div>
                    </div>
                    <div class="experience-date">${exp.date}</div>
                    <div class="experience-text">${exp.text}</div>
                `;
                
                experiencesList.appendChild(expElement);
            });
        }

        // Load sample experiences if none exist
        function loadSampleExperiences() {
            if (experiences.length === 0) {
                const sampleExperiences = [
                    {
                        id: 1,
                        name: 'Alex Johnson',
                        email: '',
                        type: 'good',
                        text: 'WhyUnified helped streamline our e-commerce operations significantly. The dashboard is intuitive and their support team was responsive.',
                        date: 'Mar 15, 2024, 10:30 AM'
                    },
                    {
                        id: 2,
                        name: 'Sam Rivera',
                        email: '',
                        type: 'bad',
                        text: 'Had issues with integration that took too long to resolve. The initial setup was not as smooth as expected.',
                        date: 'Feb 28, 2024, 03:45 PM'
                    },
                    {
                        id: 3,
                        name: 'Taylor Smith',
                        email: '',
                        type: 'good',
                        text: 'Great platform for managing multiple storefronts. The analytics feature alone has helped increase our sales by 15%.',
                        date: 'Apr 2, 2024, 09:15 AM'
                    }
                ];
                
                experiences = sampleExperiences;
                saveExperiences();
            }
        }

        // Initialize the page
        function init() {
            loadSampleExperiences();
            displayExperiences();
        }

        // Start the application
        init();
    </script>
